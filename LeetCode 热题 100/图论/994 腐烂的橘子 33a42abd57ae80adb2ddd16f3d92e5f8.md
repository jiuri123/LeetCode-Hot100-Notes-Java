# 994. 腐烂的橘子

[https://leetcode.cn/problems/rotting-oranges/description/?envType=study-plan-v2&envId=top-100-liked](https://leetcode.cn/problems/rotting-oranges/description/?envType=study-plan-v2&envId=top-100-liked)

```java
class Solution {
    public int orangesRotting(int[][] grid) {
        if (grid == null || grid.length == 0) return 0;
        
        int rows = grid.length;
        int cols = grid[0].length;
        Deque<int[]> queue = new ArrayDeque<>();
        int freshCount = 0;

        // 1. 统计新鲜橘子的数量，并将所有初始腐烂的橘子入队
        for (int r = 0; r < rows; r++) {
            for (int c = 0; c < cols; c++) {
                if (grid[r][c] == 2) {
                    queue.offer(new int[]{r, c});
                } else if (grid[r][c] == 1) {
                    freshCount++;
                }
            }
        }

        // 如果一开始就没有新鲜橘子，直接返回 0
        if (freshCount == 0) return 0;

        int minutes = 0;
        int[][] directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}}; // 上下左右四个方向

        // 2. 开始 BFS 扩散
        while (!queue.isEmpty()) {
            int size = queue.size();
            boolean isRottedInThisLayer = false;

            for (int i = 0; i < size; i++) {
                int[] curr = queue.poll();
                for (int[] d : directions) {
                    int nextR = curr[0] + d[0];
                    int nextC = curr[1] + d[1];

                    // 检查边界及是否为新鲜橘子
                    if (nextR >= 0 && nextR < rows && nextC >= 0 && nextC < cols && grid[nextR][nextC] == 1) {
                        grid[nextR][nextC] = 2; // 变腐烂
                        queue.offer(new int[]{nextR, nextC});
                        freshCount--;
                        isRottedInThisLayer = true;
                    }
                }
            }
            
            // 只有当这一层确实感染了新橘子时，时间才加 1
            if (isRottedInThisLayer) minutes++;
        }

        // 3. 最后检查是否还有无法被感染的新鲜橘子
        return freshCount == 0 ? minutes : -1;
    }
}
```
