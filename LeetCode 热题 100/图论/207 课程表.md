# 207. 课程表

[https://leetcode.cn/problems/course-schedule/description/?envType=study-plan-v2&envId=top-100-liked](https://leetcode.cn/problems/course-schedule/description/?envType=study-plan-v2&envId=top-100-liked)

```java
class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        // 构建邻接表和入度数组
        List<List<Integer>> adj = new ArrayList<>();
        for (int i = 0; i < numCourses; i++) { // 注意需要初始化
            adj.add(new ArrayList<>());
        }
        int[] indegree = new int[numCourses];
        for (int[] cp : prerequisites) {
            // 维护一个后续课程数组
            // cp[1] → cp[0]……
            adj.get(cp[1]).add(cp[0]);
            indegree[cp[0]]++;
        }

        // 将所有入度为 0 的节点入队
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < numCourses; i++) {
            if (indegree[i] == 0) {
                queue.offer(i);
            }
        }

        int count = 0; // 记录已修完的课程数
        while (!queue.isEmpty()) {
            // 开始学习无先修课程的课程
            int curr = queue.poll();
            count++;

            // 遍历当前课程的所有后续课程
            for (int next : adj.get(curr)) {
                indegree[next]--;
                // 如果后续课程的入度变为 0，说明其所有先修课已全部修完
                if (indegree[next] == 0) {
                    queue.offer(next); // 加入到待学队列
                }
            }
        }

        // 如果修完的课等于总课数，说明没有环
        return count == numCourses;
    }
}
```
