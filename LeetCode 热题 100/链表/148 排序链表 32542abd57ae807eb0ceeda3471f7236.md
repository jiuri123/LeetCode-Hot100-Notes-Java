# 148. 排序链表

[https://leetcode.cn/problems/sort-list/description/?envType=study-plan-v2&envId=top-100-liked](https://leetcode.cn/problems/sort-list/description/?envType=study-plan-v2&envId=top-100-liked)

## 方法一：拆分+合并的归并排序（面试推荐）

使用归并排序，首先将链表不断的从中间进行划分，直到每个子链表只剩下至多一个节点，最后再分别对每两个子链表进行合并。划分的操作可以使用快慢指针法找到中间节点之后，直接取中间节点的next指针为null即可。合并操作可以通过设置虚拟头结点+尾插法的方式进行合并。代码如下：

```java
// 方法一：拆分 + 合并（自顶向下递归，面试推荐）
// 时间复杂度：O(nlogn)
// 空间复杂度：O(logn) -> 主要来源于递归调用的栈空间
class Solution {
    public ListNode sortList(ListNode head) {
        // 【递归终止条件】：如果链表为空，或者只有一个节点，说明已经是有序的了，直接返回
        if (head == null || head.next == null) {
            return head;
        }

        // 1. 【寻找中点】：利用快慢指针法
        // slow 走一步，fast 走两步，当 fast 到达末尾时，slow 恰好在中间
        ListNode slow = head;
        ListNode fast = head.next; // 注意这里 fast 从 head.next 开始，是为了在偶数节点时停在左半部分末尾
        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;
        }

        // 2. 【切断链表】：找到中点后，将链表一分为二
        ListNode backHead = slow.next; // 右半部分的头节点
        slow.next = null;              // 断开连接，此时原链表 head 变成了左半部分

        // 3. 【递归拆分】：继续对左半部分和右半部分进行同样的排序操作
        ListNode left = sortList(head);     // 递归处理左边
        ListNode right = sortList(backHead); // 递归处理右边

        // 4. 【合并有序链表】：将两个已经排好序的子链表合并成一个大链表
        // 创建一个虚拟头节点 dummy，方便统一处理合并逻辑
        ListNode mergeHead = new ListNode(-1, null);
        ListNode mergeTail = mergeHead; // 用于指向合并后链表的当前末尾节点

        while (left != null && right != null) {
            // 比较左右两个链表当前的节点值，谁小就把谁接在后面
            if (left.val <= right.val) {
                mergeTail.next = left;
                left = left.next; // 左边指针后移
            } else {
                mergeTail.next = right;
                right = right.next; // 右边指针后移
            }
            mergeTail = mergeTail.next; // 合并后的末尾指针后移
        }

        // 5. 【收尾工作】：如果其中一个链表已经走完了，直接把剩下的另一个链表接上去
        mergeTail.next = (left == null ? right : left);

        // 返回合并后的真实头节点（虚拟头节点的下一个）
        return mergeHead.next;
    }
}
```

## 方法二：自底向上的迭代递归排序（加分项）

注：这个方法没有动手写，只是看了一遍题解，以后有时间再自己写一遍

```java
// 方法二：迭代法归并排序，加分项
// 时间复杂度：O(nlogn)
// 空间复杂度：O(1)
class Solution {
    public ListNode sortList(ListNode head) {
        // 1. 特判：如果链表为空，直接返回
        if (head == null) {
            return head;
        }

        // 2. 获取链表的总长度，用于后续控制循环次数
        int length = 0;
        ListNode node = head;
        while (node != null) {
            length++;
            node = node.next;
        }

        // 3. 引入虚拟头节点，方便处理排序后头节点的变化
        ListNode dummyHead = new ListNode(0, head);

        // 4. 自底向上归并：步长 subLength 从 1 开始，每次翻倍（1, 2, 4, 8...）
        // 每一次循环，都会完成全链表范围内、以 subLength 为单位的局部排序合并
        for (int subLength = 1; subLength < length; subLength <<= 1) {
            // prev 记录上一次合并完的末尾，初始指向虚拟头
            // curr 记录当前正在处理的合并位置
            ListNode prev = dummyHead, curr = dummyHead.next;

            while (curr != null) {
                // --- 找到第一个子链表 head1 ---
                ListNode head1 = curr;
                // 向后走 subLength - 1 步，到达 head1 的末尾
                for (int i = 1; i < subLength && curr.next != null; i++) {
                    curr = curr.next;
                }

                // --- 找到第二个子链表 head2 ---
                ListNode head2 = curr.next; 
                curr.next = null; // 【关键】切断 head1 和 head2 的联系
                curr = head2;     // curr 移到 head2 的起点
                
                // 向后走 subLength - 1 步，到达 head2 的末尾
                // 注意这里要判断 curr != null，因为最后一部分可能不够 subLength 个
                for (int i = 1; i < subLength && curr != null && curr.next != null; i++) {
                    curr = curr.next;
                }

                // --- 保存剩余链表的起点 ---
                ListNode next = null;
                if (curr != null) {
                    next = curr.next; // 记录下一次合并的起始位置
                    curr.next = null; // 【关键】切断 head2 和剩余链表的联系
                }

                // --- 合并 head1 和 head2 ---
                ListNode merged = merge(head1, head2);
                
                // 将合并后的有序子链表接到已排序部分的后面
                prev.next = merged;
                
                // 将 prev 移动到当前合并后的末尾，为下一次接龙做准备
                while (prev.next != null) {
                    prev = prev.next;
                }
                
                // 移动到下一组需要合并的位置
                curr = next;
            }
        }
        return dummyHead.next;
    }

    /**
     * 辅助函数：合并两个有序链表
     * 这部分逻辑和 21. 合并两个有序链表 是一样的
     */
    public ListNode merge(ListNode head1, ListNode head2) {
        ListNode dummyHead = new ListNode(0);
        ListNode temp = dummyHead, temp1 = head1, temp2 = head2;
        
        while (temp1 != null && temp2 != null) {
            if (temp1.val <= temp2.val) {
                temp.next = temp1;
                temp1 = temp1.next;
            } else {
                temp.next = temp2;
                temp2 = temp2.next;
            }
            temp = temp.next;
        }
        
        // 如果其中一个链表还没走完，直接把剩下的接在后面
        if (temp1 != null) {
            temp.next = temp1;
        } else if (temp2 != null) {
            temp.next = temp2;
        }
        return dummyHead.next;
    }
}
```
