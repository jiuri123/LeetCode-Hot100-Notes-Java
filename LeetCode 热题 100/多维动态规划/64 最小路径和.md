# 64. 最小路径和

[https://leetcode.cn/problems/minimum-path-sum/description/?envType=study-plan-v2&envId=top-100-liked](https://leetcode.cn/problems/minimum-path-sum/description/?envType=study-plan-v2&envId=top-100-liked)

## 方法一：二维DP

```java
class Solution {
    public int minPathSum(int[][] grid) {
        if (grid == null || grid.length == 0 || grid[0].length == 0) {
            return 0;
        }
        int rows = grid.length, columns = grid[0].length;
        int[][] dp = new int[rows][columns];
        dp[0][0] = grid[0][0];
        for (int i = 1; i < rows; i++) {
            dp[i][0] = dp[i - 1][0] + grid[i][0];
        }
        for (int j = 1; j < columns; j++) {
            dp[0][j] = dp[0][j - 1] + grid[0][j];
        }
        for (int i = 1; i < rows; i++) {
            for (int j = 1; j < columns; j++) {
                dp[i][j] = Math.min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];
            }
        }
        return dp[rows - 1][columns - 1];
    }
}
```

## 方法二：原地修改法（面试推荐）

```java
class Solution {
    public int minPathSum(int[][] grid) {
        if (grid == null || grid.length == 0 || grid[0].length == 0) {
            return 0;
        }

        int rows = grid.length;
        int cols = grid[0].length;

        // 遍历整个网格
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (i == 0 && j == 0) {
                    // 起点，不做处理
                    continue;
                } else if (i == 0) {
                    // 第一行：只能从左边走过来
                    grid[i][j] += grid[i][j - 1];
                } else if (j == 0) {
                    // 第一列：只能从上边走过来
                    grid[i][j] += grid[i - 1][j];
                } else {
                    // 常规位置：取上方和左方的最小值
                    grid[i][j] += Math.min(grid[i - 1][j], grid[i][j - 1]);
                }
            }
        }

        // 最终结果存储在右下角
        return grid[rows - 1][cols - 1];
    }
}
```

## 方法三：一维DP（面试进阶）

```java
public int minPathSum(int[][] grid) {
    int cols = grid[0].length;
    int[] dp = new int[cols];
    
    for (int i = 0; i < grid.length; i++) {
        for (int j = 0; j < cols; j++) {
            if (i == 0 && j == 0) dp[j] = grid[i][j]; // 起点
            else if (i == 0) dp[j] = dp[j - 1] + grid[i][j]; // 第一行
            else if (j == 0) dp[j] = dp[j] + grid[i][j]; // 第一列（dp[j]此时存的是上一行的数据）
            else dp[j] = Math.min(dp[j], dp[j - 1]) + grid[i][j];
        }
    }
    return dp[cols - 1];
}
```
