# 4. 寻找两个正序数组的中位数

[https://leetcode.cn/problems/median-of-two-sorted-arrays/description/?envType=study-plan-v2&envId=top-100-liked](https://leetcode.cn/problems/median-of-two-sorted-arrays/description/?envType=study-plan-v2&envId=top-100-liked)

```java
class Solution {
    public double findMedianSortedArrays(int[] nums1, int[] nums2) {
        // 策略：始终对较短的数组进行二分来查找切割位置。
        // 这样做一是为了快，二是保证计算 j = (m+n+1)/2 - i 时，j 永远不会是负数。
        if (nums1.length > nums2.length) {
            return findMedianSortedArrays(nums2, nums1);
        }

        int m = nums1.length;
        int n = nums2.length;
        
        // left 和 right 定义的是 nums1 里的“切割位置”，范围从 0 到 m
        int left = 0;
        int right = m;
        
        // 开始二分查找切割点
        while (left <= right) {
            // i 是在 nums1 里的切刀位置，j 是在 nums2 里的切刀位置
            int i = left + (right - left) / 2;
            // (m + n + 1) / 2 保证了左半部分的总数 = 右半部分（或多一个）
            int j = (m + n + 1) / 2 - i;

            // --- 处理边界情况（哨兵值） ---
            // 如果 i == 0，说明 nums1 左边没数了，给它一个最小整数，不影响后面的比较
            int leftMax1 = (i == 0) ? Integer.MIN_VALUE : nums1[i - 1];
            // 如果 i == m，说明 nums1 右边没数了，给它一个最大整数
            int rightMin1 = (i == m) ? Integer.MAX_VALUE : nums1[i];

            // nums2 同理
            int leftMax2 = (j == 0) ? Integer.MIN_VALUE : nums2[j - 1];
            int rightMin2 = (j == n) ? Integer.MAX_VALUE : nums2[j];

            // --- 核心判断逻辑 ---
            // 完美的切割点需要满足：左边所有的数 <= 右边所有的数
            // 因为数组本身有序，我们只需交叉检查：
            if (leftMax1 <= rightMin2 && leftMax2 <= rightMin1) {
                // 情况一：总长度是奇数
                if ((m + n) % 2 == 1) {
                    // 因为我们计算 j 时多加了1，中位数就在左半部分的边缘（两个左侧最大值中的较大者）
                    return Math.max(leftMax1, leftMax2);
                } 
                // 情况二：总长度是偶数
                else {
                    // 中位数是左侧最大和右侧最小的平均值
                    return (Math.max(leftMax1, leftMax2) + Math.min(rightMin1, rightMin2)) / 2.0;
                }
            } 
            // --- 调整二分范围 ---
            // 如果 nums1 左边太大了，切刀需要往左移
            else if (leftMax1 > rightMin2) {
                right = i - 1;
            } 
            // 如果 nums1 左边太小了，切刀需要往右移
            else {
                left = i + 1;
            }
        }
        return 0.0;
    }
}
```
